 121|export function suggestCommandForAction(

 124|  commands: CommandDef[],
 125|  context?: { entityName: string; boundDevice?: Pick<Device, "name" | "type">; devices?: Device[] },

 127|  if (commands.length === 0) return undefined;
 128|
 129|  const start = isStartAction(actionId, actionLabel);
 130|  const stop = isStopAction(actionId, actionLabel);
 131|  const prefix = start ? "start" : stop ? "stop" : "";
 132|  const entityName = context?.entityName ?? "";
 133|  const devices = context?.devices ?? [];
 134|  const preferredName = entityName ? preferTargetName(entityName, devices) : undefined;
 135|
 136|  const exact = commands.find((c) => c.id === actionId || c.label === actionLabel);
 137|  if (exact && !isBroadcastCommand(exact)) {
 138|    const params = targetParamsForCommand(exact, entityName, devices, preferredName, context?.boundDevice);
 139|    return { command: exact, params };
 140|  }
 141|
 142|  if (prefix) {
 143|    const plain = commands.find((c) => c.id.toLowerCase() === prefix || c.label === (start ? "启动" : "停止"));
 144|    if (plain && !isBroadcastCommand(plain)) {
 145|      const params = targetParamsForCommand(plain, entityName, devices, preferredName, context?.boundDevice);
 146|      return { command: plain, params };
 147|    }

 149|    const prefixed = commands
 150|      .filter((c) => {
 151|        const id = c.id.toLowerCase();

 153|      })
 154|      .map((command) => {
 155|        const target = commandTarget(command) || command.label;
 156|        const score = Math.max(

 162|      .filter((item) => item.score >= MATCH_THRESHOLD)

 167|      return { command: topPrefixed.command };
